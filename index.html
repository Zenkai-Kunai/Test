// Example: Connect to a live real-time WebSocket feed for XAUUSD const ws = new WebSocket('wss://ws.twelvedata.com/v1/quotes/price?apikey=YOUR_API_KEY'); ws.onopen = () => { // Subscribe to real-time Gold (XAU/USD) ticks ws.send(JSON.stringify({ "action": "subscribe", "params": { "symbols": "XAU/USD" } })); }; ws.onmessage = (event) => { const data = JSON.parse(event.data); if (data.event === 'price') { const liveGoldPrice = data.price; // Update live candle array with true real-time price updateCandleData(liveGoldPrice); } };
